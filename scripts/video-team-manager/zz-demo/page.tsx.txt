export default function Demo() {
  const [s, setS] = useState('');
  useEffect(() => { queueMicrotask(() => setS(D.q('s'))); document.documentElement.setAttribute('data-theme', 'light'); }, []);
  if (!s) return null;
  const inhalt = (() => {
    if (s === 'dash') return <TeamManager ansicht="dashboard" />;
    if (s === 'termine') return <TeamManager ansicht="termine" />;
    if (s === 'kasse') return <KassePage />;
    if (s === 'gegner') return <GegnerSeite />;
    if (s === 'premium') return <PremiumSeite />;
    if (s === 'kalender') return <MemberShell title="Mein Spielplan"><div style={{ maxWidth: 640 }}><KalenderAbo teamId={D.TEAM} teamName={team27(D.TEAM)!.name} /></div></MemberShell>;
    if (s === 'bericht') return <BerichtDemo />;
    if (s === 'spiel') {
      const b = alleBegegnungen27().find(x => x.key === D.SPIEL)!;
      const ort = venue27(team27(b.home)?.venueId ?? null);
      return <div style={{ background: 'var(--th-bg-page)', color: 'var(--th-text-strong)', minHeight: '100vh' }}>
        <DesktopHeader activeHref="/mein-bereich" />
        <div className="mdu-section-pad" style={{ maxWidth: 640, margin: '0 auto', padding: '24px 16px 80px' }}>
          <SpieltagAnsicht b={b} teamId={D.TEAM} teamName={team27(D.TEAM)!.name} gegner={team27(b.away)!.name}
            liga={findLiga27(b.liga)?.name ?? ''} ort={ort ? { name: ort.name, address: ort.address ?? null } : null} />
        </div></div>;
    }
    return null;
  })();
  return <>{inhalt}<GlockeImKopf />{D.q('push') === '1' && <PushAbfrageDemo />}</>;
}

/** Die echte Glocke im Kopf ausblenden und die Demo-Glocke an ihrer Stelle einhängen. */
function GlockeImKopf() {
  const [ziele, setZiele] = useState<HTMLElement[]>([]);
  useEffect(() => {
    const iv = setInterval(() => {
      const echte = [...document.querySelectorAll<HTMLElement>('button[aria-label^="Benachrichtigungen"]')].filter(b => !b.closest('.demo-glocke'));
      if (!echte.length) return;
      clearInterval(iv);
      setZiele(echte.map(echt => {
        echt.style.display = 'none';
        const platz = document.createElement('span'); platz.className = 'demo-glocke'; platz.style.display = 'contents';
        echt.parentElement!.insertBefore(platz, echt); return platz;
      }));
    }, 150);
    return () => clearInterval(iv);
  }, []);
  return <>{ziele.map((z, i) => createPortal(<NotificationBell />, z, 'g' + i))}</>;
}

// ── Spielbericht: Abschnitt „Aufstellung“ mit dem Übernehmen-Knopf (Markup aus app/mein-bereich/spielberichte) ──
function BerichtDemo() {
  const leer = (side: 'home' | 'guest'): ReportPlayer[] => Array.from({ length: 8 }, (_, i) => ({ side, slot: i + 1, pass_no: null, name: '', player_id: null, points: 0 }));
  const [home, setHome] = useState<ReportPlayer[]>(leer('home'));
  const [guest, setGuest] = useState<ReportPlayer[]>(leer('guest'));
  const [msg, setMsg] = useState<string | null>(null);
  const lineup = D.DETAILS[D.SPIEL].lineup;
  const uebernehmen = () => {
    setHome(arr => arr.map(p => { const k = D.KADER.find(x => x.id === lineup[p.slot - 1]); return k ? { ...p, name: k.name, player_id: k.id } : p; }));
    setMsg(`Aufstellung übernommen – Heim: ${lineup.filter(Boolean).length} Spieler · Gast: nichts hinterlegt. Bitte kurz prüfen.`);
  };
  useEffect(() => { if (D.q('voll') === '1') queueMicrotask(uebernehmen); }, []); // eslint-disable-line react-hooks/exhaustive-deps
  const gast = alleBegegnungen27().find(x => x.key === D.SPIEL)!.away;
  const gastKader: KaderOption[] = ['Rudi Rakete', 'Moni Maximum', 'Bert Bogen', 'Lotte Leg', 'Fritz Finish'].map((name, i) => ({ id: 'g' + i, name, isCaptain: i === 0, passNo: null }));
  void gast;
  return (
    <MemberShell title="Spielbericht erfassen">
      <div style={{ display: 'flex', flexDirection: 'column', gap: 16 }}>
        {msg && <div role="status" style={{ padding: '10px 14px', borderRadius: 8, fontFamily: 'var(--font-manrope)', fontSize: 13, background: 'rgba(34,197,94,0.10)', border: '1px solid rgba(34,197,94,0.35)', color: 'var(--th-win)' }}>{msg}</div>}
        <Section title="Aufstellung (H1–H8 / G1–G8 · Position 1–4 startet, 5–8 Wechsel)">
          <div style={{ display: 'flex', alignItems: 'center', gap: 10, flexWrap: 'wrap', marginBottom: 12, padding: '10px 12px', borderRadius: 10, background: 'var(--th-accent-a07)', border: '1px solid var(--th-accent-a25)' }}>
            <span style={{ flex: '1 1 220px', fontFamily: 'var(--font-manrope)', fontSize: 12.5, color: 'var(--th-text-body)', lineHeight: 1.45 }}>
              <strong style={{ color: 'var(--th-text-strong)' }}>Aufstellung aus dem Team Manager:</strong>{' '}
              Heim {lineup.filter(Boolean).length} Spieler · Gast – nichts hinterlegt
            </span>
            <button id="uebernehmen" type="button" onClick={uebernehmen} style={{ padding: '8px 14px', borderRadius: 8, cursor: 'pointer', fontFamily: 'var(--font-manrope)', fontWeight: 800, fontSize: 12.5, border: '1.5px solid var(--th-accent)', background: 'var(--th-accent)', color: '#fff' }}>Übernehmen</button>
          </div>
          <div style={{ display: 'grid', gridTemplateColumns: '1fr 1fr', gap: 16 }}>
            <Lineup label="Heim" players={home} setPlayers={setHome} prefix="H" options={D.KADER} teamChosen />
            <Lineup label="Gast" players={guest} setPlayers={setGuest} prefix="G" options={gastKader} teamChosen />
          </div>
        </Section>
      </div>
    </MemberShell>
  );
}

function Lineup({ label, players, setPlayers, prefix, options, teamChosen }: {
  label: string; players: ReportPlayer[];
  setPlayers: (f: (a: ReportPlayer[]) => ReportPlayer[]) => void;
  prefix: string; options: KaderOption[]; teamChosen: boolean;
}) {
  function pick(slot: number, id: string) {
    const opt = options.find(o => o.id === id);
    setPlayers(arr => arr.map(x => x.slot === slot ? { ...x, player_id: id || null, name: opt ? opt.name : '', pass_no: opt?.passNo ?? x.pass_no } : x));
  }
  return (
    <div>
      <div style={{ fontFamily: 'var(--font-manrope)', fontWeight: 800, fontSize: 12, color: 'var(--th-accent)', marginBottom: 6 }}>{label}</div>
      {players.map(p => {
        const usedByOthers = new Set(players.filter(x => x.slot !== p.slot && x.player_id).map(x => x.player_id));
        return (
          <div key={p.slot} style={{ display: 'flex', alignItems: 'center', gap: 6, marginBottom: 6 }}>
            <span style={{ width: 26, fontFamily: 'var(--font-jetbrains-mono)', fontSize: 12, color: p.slot <= 4 ? 'var(--th-text-body)' : 'var(--th-text-faint)' }}>{prefix}{p.slot}</span>
            <select value={p.player_id ?? ''} onChange={e => pick(p.slot, e.target.value)} disabled={!teamChosen} style={{ ...input, flex: 1, padding: '7px 9px', opacity: teamChosen ? 1 : 0.6 }}>
              <option value="">{p.slot <= 4 ? '— Spieler wählen * —' : '— Wechsel (optional) —'}</option>
              {options.map(o => <option key={o.id} value={o.id} disabled={usedByOthers.has(o.id)}>{o.name}{o.passNo ? ` · ${o.passNo}` : ''}{o.isCaptain ? ' (C)' : ''}{usedByOthers.has(o.id) ? ' — bereits gewählt' : ''}</option>)}
            </select>
          </div>
        );
      })}
    </div>
  );
}

function Section({ title, children }: { title: string; children: React.ReactNode }) {
  return (
    <div style={{ background: 'var(--th-bg-card)', border: '1px solid var(--th-line-6)', borderRadius: 14, padding: '18px 20px', display: 'flex', flexDirection: 'column', gap: 12 }}>
      <div style={{ fontFamily: 'var(--font-manrope)', fontWeight: 800, fontSize: 11, letterSpacing: '0.14em', color: 'var(--th-accent)', textTransform: 'uppercase' }}>{title}</div>
      {children}
    </div>
  );
}

const input: React.CSSProperties = {
  width: '100%', padding: '9px 12px', background: 'var(--th-bg-header)', border: '1px solid var(--th-line-10)',
  borderRadius: 8, color: 'var(--th-text-strong)', fontFamily: 'var(--font-manrope)', fontSize: 14, outline: 'none',
};
